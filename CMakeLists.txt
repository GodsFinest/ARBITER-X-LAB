cmake_minimum_required(VERSION 3.24)
project(arbiter_x_lab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ARBITER_WARNINGS_AS_ERRORS "Treat warnings as errors" OFF)
option(ARBITER_ENABLE_SANITIZERS "Enable ASan/UBSan" OFF)
option(ARBITER_LAB_BUILD_BENCHMARKS "Build research benchmark" ON)
set(ARBITER_X_SOURCE_DIR "" CACHE PATH "Explicit core source checkout")
if(TARGET ArbiterX::core)
  message(FATAL_ERROR "Duplicate ArbiterX provider; configure LAB with exactly one provider")
endif()
if(ARBITER_X_SOURCE_DIR)
  if(NOT EXISTS "${ARBITER_X_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Invalid ARBITER_X_SOURCE_DIR")
  endif()
  add_subdirectory("${ARBITER_X_SOURCE_DIR}" _deps/arbiter-x)
else()
  find_package(ArbiterX 1.1.0...<1.2.0 CONFIG REQUIRED)
endif()
get_target_property(core_version ArbiterX::core ARBITER_X_VERSION)
get_target_property(core_api ArbiterX::core ARBITER_X_API_VERSION)
if(NOT core_version VERSION_GREATER_EQUAL 1.1.0 OR NOT core_version VERSION_LESS 1.2.0 OR NOT core_api STREQUAL "1")
  message(FATAL_ERROR "Unsupported ArbiterX version/API: ${core_version}/${core_api}; require 1.1.x/API 1")
endif()
function(lab_configure target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
    if(ARBITER_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(ARBITER_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
    if(ARBITER_ENABLE_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endfunction()
add_library(arbiter_lab STATIC
  src/data/data.cpp src/research/features.cpp src/research/labels.cpp
  src/research/quant_model.cpp src/research/execution_research.cpp
  src/research/pipeline.cpp src/research/research.cpp src/experiment/experiment.cpp)
add_library(ArbiterXLab::research ALIAS arbiter_lab)
target_include_directories(arbiter_lab PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(arbiter_lab PUBLIC ArbiterX::core)
target_compile_features(arbiter_lab PUBLIC cxx_std_23)
lab_configure(arbiter_lab)
add_executable(arbiter-lab src/cli/main.cpp src/cli/research_cli.cpp src/cli/experiment_cli.cpp)
target_link_libraries(arbiter-lab PRIVATE arbiter_lab)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_custom_target(lab_build_info
  COMMAND ${CMAKE_COMMAND} "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}" "-DVERSION=${PROJECT_VERSION}" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/generated/arbiter_build_info_config.hpp" -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/GenerateBuildInfo.cmake
  BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/generated/arbiter_build_info_config.hpp
  VERBATIM)
add_dependencies(arbiter-lab lab_build_info)
target_include_directories(arbiter-lab PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/CORE_REVISION")
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/CORE_REVISION tested_core_revision)
string(STRIP "${tested_core_revision}" tested_core_revision)
string(LENGTH "${tested_core_revision}" tested_core_length)
if(NOT tested_core_length EQUAL 40 OR NOT tested_core_revision MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "CORE_REVISION must contain one full Git SHA")
endif()
target_compile_definitions(arbiter-lab PRIVATE AX_TESTED_CORE_REVISION="${tested_core_revision}")
lab_configure(arbiter-lab)
if(ARBITER_LAB_BUILD_BENCHMARKS)
  add_executable(arbiter_lab_bench benchmarks/bench_main.cpp)
  target_link_libraries(arbiter_lab_bench PRIVATE arbiter_lab)
  lab_configure(arbiter_lab_bench)
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(test_name IN ITEMS test_quant test_experiment)
    add_executable(${test_name} tests/${test_name}.cpp)
    target_link_libraries(${test_name} PRIVATE arbiter_lab)
    target_compile_definitions(${test_name} PRIVATE AX_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    lab_configure(${test_name})
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME lab_python_features COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/research/reference_features.py --compare $<TARGET_FILE:arbiter-lab>)
    set_tests_properties(lab_python_features PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  endif()
  foreach(rejection IN ITEMS version api)
    add_test(NAME lab_reject_${rejection} COMMAND ${CMAKE_COMMAND} "-DCASE=${rejection}" "-DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}" "-DLAB_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}" -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cmake/reject_dependency.cmake)
  endforeach()
  foreach(invalid IN ITEMS seed queue incomplete threshold)
    if(invalid STREQUAL "seed")
      set(invalid_args "research;evaluate;--seed;-1")
    elseif(invalid STREQUAL "queue")
      set(invalid_args "research;backtest;--queue;bogus")
    elseif(invalid STREQUAL "threshold")
      set(invalid_args "research;run;EXP-001;--threshold;NaN")
    else()
      set(invalid_args "research;features;--out")
    endif()
    add_test(NAME lab_invalid_${invalid} COMMAND ${CMAKE_COMMAND} "-DPROGRAM=$<TARGET_FILE:arbiter-lab>" "-DARGS=${invalid_args}" -DEXPECTED=2 -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cmake/expect_exit.cmake)
  endforeach()
  add_test(NAME lab_help COMMAND arbiter-lab --help)
  add_test(NAME lab_unknown COMMAND arbiter-lab unknown)
  set_tests_properties(lab_unknown PROPERTIES WILL_FAIL TRUE)
endif()
